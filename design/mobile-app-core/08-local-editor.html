import { z } from "zod";import { auth } from "@/lib/auth";
const loginSchema = z.object({
  email: z.string().email("Email inválido"),
  password: z.string().min(8, "Mínimo 8 caracteres"),
});export async function login(input: unknown) {  const parsed = loginSchema.safeParse(input);  if (!parsed.success) {    const errors = parsed.error.flatten().fieldErrors;    return { ok: false, errors };  }  const { email, password } = parsed.data;  return auth.signIn(email, password);}
